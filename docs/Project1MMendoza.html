Parsing Variable Values

The function “variable_get” is a function designed to take the value of “YEAR” as a character input. It will return the variable list for the specified year (“2021”, “2022”, “2023”, “2024”) by creating the URL from the year associated with the .json file, parse the results into a datalist, and then will match and output the required variables for the specified year, and save the list as an .rds file in the “Data” subfolder.

# Function to Parse Variable Values
variable_get <- function( YEAR) {
  
  # 1) Create jsonURL based on year selected
  jsonURL <- str_c("https://api.census.gov/data/", YEAR, 
                   "/acs/acs1/pums/variables.json")

  # 2) Use get() to call unparsed data
  unparsed_data <- httr::GET(jsonURL)
  str(unparsed_data, max.level = 1)
  
  # 3) parse returned information for the actual article information
  raw_json <- content(unparsed_data, as = "text")
  raw_data <- fromJSON(raw_json)
  parsed_data <- raw_data$variables
  
  # 4) Filter only necessary variables
    # 4.1) If  condition to alter needed variables based on year
    if (!(YEAR %in% c("2021", "2022", "2023", "2024"))) {
      stop("ERROR: input Variable YEAR is not in list of 2021, 2022, 2023, 2024")
    }
    else {
    if (YEAR %in% c("2021", "2022")) {
    varlist <- c("AGEP", "GASP", "GRPIP", "JWAP", "JWDP", "JWMNP", 
                 "PWGTP", "FER", "HHL", "SCH", "SCHL", "SEX", 
                 "REGION", "DIVISION", "ST")
    }
    else if (YEAR %in% c("2023", "2024")) {
    varlist <- c("AGEP", "GASP", "GRPIP", "JWAP", "JWDP", "JWMNP",
                 "PWGTP", "FER", "HHL", "SCH", "SCHL", "SEX", 
                 "REGION", "DIVISION", "STATE")
        
    }
    }
    # 4.2) create a list of variables that are both requested and present
    targets <- intersect(varlist, names(parsed_data))
    
    # 4.3) select matching vars from data. 
    output <- parsed_data[targets]
  
  
  # 5) save .RDS file to Datafolder
      # 5.1) Create unique file name per year to save to data folder
        outfile <- str_c("Data/", YEAR, ".rds")
        
      # 5.2) Save output as RDS
        saveRDS(output, file = outfile)
        
  #this is just for QC, print dataframe for review
  return(output)
}

#HELPER FUNCTION 2: DATE TIME (Helped from LLMs as per guidelines) time interval #strings and returns the midpoint time
parse_pums_time_midpoint <- function(time_code, var_name, all_levels) {
  # 1. Handle missing/invalid data inputs safely
  if (is.na(time_code) || time_code == "" || time_code == "N/A") return(hms(NA))
  
  # 2. Grab the actual descriptive string label mapped to the numeric code
  # Example: Code "020" maps to "06:00 a.m. to 06:29 a.m."
  time_str <- all_levels[[var_name]]$values$item[[as.character(time_code)]]
  
  if (is.null(time_str) || is.na(time_str)) return(hms(NA))
  
  # 3. Extract the two HH:MM timestamps with their AM/PM flags
  times <- str_extract_all(time_str, "\\d{2}:\\d{2}\\s*(a\\.m\\.|p\\.m\\.|am|pm)")[[1]]
  if (length(times) < 2) return(hms(NA)) 
  
  # 4. Extract hour/minute parts using standard lubridate structures
  t1 <- hm(str_extract(times[1], "\\d{2}:\\d{2}"))
  t2 <- hm(str_extract(times[2], "\\d{2}:\\d{2}"))
  
  # 5. Apply standardized 24-hour PM adjustments
  if (str_detect(tolower(times[1]), "p\\.m\\.|pm") && hour(t1) < 12) t1 <- t1 + hours(12)
  if (str_detect(tolower(times[2]), "p\\.m\\.|pm") && hour(t2) < 12) t2 <- t2 + hours(12)
  if (t2 < t1) t2 <- t2 + hours(24)
  
  # 6. Compute exact midpoint duration scale
  s1 <- as.numeric(t1)
  s2 <- as.numeric(t2)
  midpoint_seconds <- (s1 + s2) / 2
  
  return(seconds_to_period(midpoint_seconds %% 86400))
}

The following code creates a helper function to combine datasets and assigning the column of “YEAR” to the tibble. it accomplishes this by assigning the year to each variable year within a datalist. it then combines the datalist together to output a single table. this datalist wil be created in the next function called “parse_pums_multi”.

# Helper function to combine combine a list of single-year dataset.
combine_pums_years <- function(data_list) {
  # Extract year tracking strings from the list names
  years <- names(data_list)
  
  # Inject the year column into each individual data frame
  for (i in seq_along(data_list)) {
    data_list[[i]] <- cbind(year = years[i], data_list[[i]])
  }
  
  # Row-bind all datasets
  combined_df <- do.call(rbind, data_list)
  
  # Structure the final output as a census tibble
  final_tibble <- as_tibble(combined_df)
  class(final_tibble) <- c("census", class(final_tibble))
  
  return(final_tibble)
}

The following function creates a function to calculate Numeric Summaries for the census Summary method. it first auto populates all numeric and categorical variables permited into num_vars and cat_vars when either is null. then it computes the weighted means and standard deviations based on the formulas provided for all numeric variables, Lastly, it computes the counts for each categorical variables and returns the results for review.

# Summarize Function
summarize.census <- function(data, 
                             num_vars = NULL, 
                             cat_vars = NULL,
                             weight = "PWGTP") {
  
  # Selects all numeric variables except specified ones if num_vars is NULL
  if (is.null(num_vars)) {
    num_vars <- data %>% 
      select(where(is.numeric)) %>% 
      select(-any_of(weight)) %>% 
      names()
  }
  
  # Selects all categorical variables except specified ones if cat_vars is NULL
  if (is.null(cat_vars)) {
    cat_vars <- data %>% 
      select(-any_of(c(num_vars, weight_col))) %>% 
      names()
  }
  
  results <- list()
  
  # Compute weighted means and standard deviations for numeric variables
  if (length(num_vars) > 0) {
    for (var in num_vars) {
      num_vec <- data[[var]]
      wt_vec  <- data[[weight]]
      
      # Sample mean formula provided
      s_mean <- sum(num_vec * wt_vec) / sum(wt_vec)
      # Sample standard deviation formula provided
      s_sd   <- sqrt(sum((num_vec^2) * wt_vec) / sum(wt_vec) - s_mean^2)
      
      results[[paste0(v, "_mean")]] <- s_mean
      results[[paste0(v, "_sd")]]   <- s_sd
    }
  }
  
  # Compute counts for categorical variables
  if (length(cat_vars) > 0) {
    for (var in cat_vars) {
      cat_counts <- data %>%
        group_by(.data[[var]]) %>%
        summarize(count = n(), .groups = "drop")
      
      results[[paste0(var, "_counts")]] <- deframe(cat_counts)
    }
  }
  
  return(results)
}
